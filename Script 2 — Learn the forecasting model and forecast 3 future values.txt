import numpy as np

# -----------------------------
# 1. Input time series
# -----------------------------
y = np.array([
    3.64973815, 2.99549717, 2.56877414, 2.9468194,
    3.63628318, 2.20029144, 3.58499797, 2.30541867,
    2.44001707, 1.91864557, 2.33640267, 0.71044036,
    1.25885197, 1.175712,   1.82684075, 1.090066,
    1.73614439, 1.84846206, 2.72661025, 3.56022295,
    3.59882147, 5.28710702, 6.03863768, 6.76454015,
    7.82283933, 8.07749541, 9.15639392, 9.62977085,
    10.6195424, 11.56870074, 11.60384656, 12.13092488,
    12.30468568, 12.41261552, 12.53969402, 12.75664092
], dtype=float)

# -----------------------------
# 2. Settings
# -----------------------------
T = len(y)
L = 6

# -----------------------------
# 3. Build page matrix
# -----------------------------
num_cols = T // L
P = y.reshape(num_cols, L).T

# -----------------------------
# 4. SVD and rank-r approximation
# -----------------------------
U, s, Vt = np.linalg.svd(P, full_matrices=True)

energy = s**2
energy_ratio = np.cumsum(energy) / np.sum(energy)
r = np.searchsorted(energy_ratio, 0.9) + 1

Sigma_r = np.zeros_like(P, dtype=float)
for i in range(r):
    Sigma_r[i, i] = s[i]

P_hat = U @ Sigma_r @ Vt

print("r =", r)
print("\nDenoised page matrix P_hat:")
print(P_hat)

# -----------------------------
# 5. Build Y and F_hat
#
# Y = last row of original page matrix P
# F_hat = first L-1 rows of denoised page matrix P_hat
# -----------------------------
Y = P[-1, :]              # shape: (num_cols,)
F_hat = P_hat[:-1, :]     # shape: (L-1, num_cols)

print("\nY (target vector):")
print(Y)

print("\nF_hat (predictor matrix):")
print(F_hat)

# -----------------------------
# 6. Solve least squares with pseudoinverse
#
# We want:
#   min_beta || Y - F_hat^T beta ||_2^2
#
# so beta = pinv(F_hat^T) @ Y
# -----------------------------
Phi = F_hat.T
beta = np.linalg.pinv(Phi) @ Y

print("\nBeta coefficients from pseudoinverse:")
for i, b in enumerate(beta, start=1):
    print(f"beta_{i} = {b:.12f}")

# -----------------------------
# 7. Forecast recursively
#
# Formula:
#   y_hat(T+1) = sum_{k=1}^{L-1} beta_k * f_hat(T+1-k)
#
# We use the denoised signal read from P_hat
# -----------------------------

# Reconstruct denoised time series from page matrix by reading columns
f_hat = P_hat.T.reshape(-1)

print("\nDenoised signal f_hat:")
print(f_hat)

# Make a working list for recursive forecasting
forecast_series = list(f_hat)

for step in range(3):
    next_val = 0.0
    for k in range(1, L):
        next_val += beta[k-1] * forecast_series[-k]
    forecast_series.append(next_val)
    print(f"Forecast y_hat(T+{step+1}) = {next_val:.12f}")

print("\nFinal 3 forecasts:")
print("y_hat(T+1) =", forecast_series[T])
print("y_hat(T+2) =", forecast_series[T+1])
print("y_hat(T+3) =", forecast_series[T+2])
